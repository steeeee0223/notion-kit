---
title: Settings Panel Presets
description: Tables for members, guests, invitations, teamspaces, plans, sessions, and connections.
---

## Installation

```package-install
pnpm add @notion-kit/settings-panel
```

## Components

---

### Sessions Table

A table to display all active sessions in the workspace.

<RegistryStory name="settings-panel-sessions-table" />

```tsx
<SessionsTable currentSessionId={currentSessionId} data={data} />
```

#### API Reference

<TypeTable
  type={{
    data: {
      type: "SessionRow[]",
      typeDescriptionLink: "/docs/blocks/settings-panel/objects#session-row",
      required: true,
      description: "All active sessions.",
    },
    currentSessionId: {
      type: "string",
      description: "Unique identifier of the current session.",
    },
    onLogout: {
      type: "(deviceName: string, token: string) => void",
      parameters: [
        {
          name: "deviceName",
          type: "string",
          description: "The name of the device being logged out.",
        },
        {
          name: "token",
          type: "string",
          description: "The session token to revoke.",
        },
      ],
      returns: "void",
      description: "Handler that is called when a session is revoked.",
    },
  }}
/>

      type: "(teamspace: TeamspaceRow) => void | Promise<void>",

      description: "Handler that is called when the teamspace is selected.",

    onUpdate: {
      type: "(data: {id: string, name?: string, icon?: IconData, description?: string, permission?: TeamspacePermission}) => void | Promise<void>",
      parameters: [
        {
          name: "data",
          type: "{id: string, name?: string, icon?: IconData, description?: string, permission?: TeamspacePermission}",
          description: "The record or fields for the operation.",
        },
      ],
      returns: "void | Promise<void>",
      description: "Handler that is called when the teamspace is updated.",
    },
    onArchive: {
      type: "(teamspace: TeamspaceRow) => void | Promise<void>",
      parameters: [
        {
          name: "teamspace",
          type: "TeamspaceRow",
          description: "The selected teamspace record.",
        },
      ],
      returns: "void | Promise<void>",
      description: "Handler that is called when the teamspace is archived.",
    },
    onLeave: {
      type: "(teamspace: TeamspaceRow) => void | Promise<void>",
      parameters: [
        {
          name: "teamspace",
          type: "TeamspaceRow",
          description: "The selected teamspace record.",
        },
      ],
      returns: "void | Promise<void>",
      description: "Handler that is called when the user leaves the teamspace.",
    },
