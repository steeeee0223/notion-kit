## API Reference

### MapClusterLayer

Creates a clustered GeoJSON source for point data and renders cluster circles, cluster counts, and unclustered point circles.

<TypeTable
  type={{
    data: {
      type: "string | FeatureCollection<Point, P>",
      required: true,
      description: "GeoJSON data or a URL to fetch GeoJSON from.",
    },
    clusterMaxZoom: {
      type: "number",
      default: "14",
      description: "Maximum zoom level for clustering.",
    },
    clusterRadius: {
      type: "number",
      default: "50",
      description: "Cluster radius in pixels.",
    },
    clusterColors: {
      type: "[string, string, string]",
      default: '["#22c55e", "#eab308", "#ef4444"]',
      description: "Colors for small, medium, large clusters.",
    },
    clusterThresholds: {
      type: "[number, number]",
      default: "[100, 750]",
      description: "Point count thresholds for cluster steps.",
    },
    pointColor: {
      type: "string",
      default: '"#3b82f6"',
      description: "Color for unclustered points.",
    },
    onPointClick: {
      type: "(feature, coordinates) => void",
      description: "Called when an unclustered point is clicked.",
    },
    onClusterClick: {
      type: "(clusterId, coordinates, pointCount) => void",
      description: "Overrides the default cluster zoom behavior.",
    },
  }}
