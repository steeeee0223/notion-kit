## API Reference

### MapArc

Builds curved `LineString` features between coordinate pairs and renders them as a MapLibre line layer with an invisible hit layer for interactions.

<TypeTable
  type={{
    data: {
      type: "MapArcDatum[]",
      typeDescriptionLink: "/docs/map/arcs#maparcdatum",
      required: true,
      description: (
        <>
          {" "}
          Arcs with <code>id</code>, <code>from</code>, and <code>to</code>{" "}
          coordinates.{" "}
        </>
      ),
    },
    id: {
      type: "string",
      description: "Optional source and layer id prefix.",
    },
    curvature: {
      type: "number",
      default: "0.2",
      description: "How far each arc bows from a straight line.",
    },
    samples: {
      type: "number",
      default: "64",
      description: "Number of samples used for the curve.",
    },
    paint: {
      type: 'LineLayerSpecification["paint"]',
      description: "MapLibre paint properties for the arc layer.",
    },
    layout: {
      type: 'LineLayerSpecification["layout"]',
      description: "MapLibre layout properties for the arc layer.",
    },
    hoverPaint: {
      type: 'LineLayerSpecification["paint"]',
      description: "Paint applied to the currently hovered arc.",
    },
    onClick: {
      type: "(event: MapArcEvent) => void",
      parameters: [
        {
          name: "event",
          type: "MapArcEvent",
          description: "The arc pointer event.",
        },
      ],
      returns: "void",
      description: "Called when an arc is clicked.",
    },
    onHover: {
      type: "(event: MapArcEvent | null) => void",
      parameters: [
        {
          name: "event",
          type: "MapArcEvent | null",
          description:
            "The arc pointer event, or null when the pointer leaves the arc.",
        },
      ],
      returns: "void",
      description: "Called when the hovered arc changes.",
    },
    interactive: {
      type: "boolean",
      default: "true",
      description: "Enables arc mouse events.",
    },
    beforeId: {
      type: "string",
      description: "Layer id to insert before for z-order control.",
    },
  }}
/>

### MapArcDatum

Describes one arc rendered by `MapArc`.

<TypeTable
  type={{
    id: {
      type: "string | number",
      required: true,
      description: "Unique arc id used for events/hover state.",
    },
    from: {
      type: "[number, number]",
      required: true,
      description: (
        <>
          {" "}
          Start coordinate as <code>[longitude, latitude]</code>.{" "}
        </>
      ),
    },
    to: {
      type: "[number, number]",
      required: true,
      description: (
        <>
          {" "}
          End coordinate as <code>[longitude, latitude]</code>.{" "}
        </>
      ),
    },
  }}
/>
